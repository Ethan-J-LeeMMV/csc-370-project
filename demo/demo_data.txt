-- queries for demo

SELECT COUNT(*) AS users FROM User;
SELECT COUNT(*) AS artists FROM Artist;
SELECT COUNT(*) AS stats_rows FROM ArtistDailyStats;

-- current roster of team 1
SELECT t.team_name, a.artist_name, r.acquired_via, r.acquired_at
FROM Rosters r
JOIN Team t   ON t.team_id   = r.team_id
JOIN Artist a ON a.artist_id = r.artist_id
WHERE t.team_id = 1 AND r.released_at IS NULL;

-- total streams per artist
SELECT a.artist_name, SUM(s.streams) AS total_streams
FROM Artist a
JOIN ArtistDailyStats s ON s.artist_id = a.artist_id
GROUP BY a.artist_id, a.artist_name
ORDER BY total_streams DESC;
